// 收到注册请求时，将人脸数据写入数据库
// 然后返回"受理成功"，5 秒后由回调任务推送识别结果
async function handle(req, res, db) {
  // 1. 校验必填参数
  const { name, imageUrl, requestId } = req.body;
  if (!name || !imageUrl) {
    return {
      code: 400,
      message: '缺少必填参数 name 或 imageUrl',
    };
  }

  // 2. 构造人脸数据
  const faceData = {
    faceId: 'face_' + Date.now(),
    name: name,
    imageUrl: imageUrl,
    requestId: requestId || '',
    metadata: req.body.metadata || {},
    createdAt: new Date().toISOString(),
  };

  // 3. 写入数据库（数据联动）
  await db.insert('face_data', faceData);

  // 4. 记录日志
  log.info('人脸注册成功', faceData);

  // 5. 返回受理结果
  return {
    code: 0,
    message: '受理成功',
    "data": {
      faceId: faceData.faceId,
      requestId: requestId,
      estimatedSeconds: 5,
    },
  };
}

// 导出供其他接口复用
module.exports = { handle };
沙箱限制：fs / net / child_process 等敏感模块被禁用。返回对象即接口响应；用 res.status(400) 可修改状态码。
